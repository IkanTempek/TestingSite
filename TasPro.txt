local function loadAS()
    if not hasFile then return end
    local ok,raw=pcall(readfile,SETTINGS_FILE); if not ok or not raw then return end
    local ok2,d=pcall(function() return Http:JSONDecode(raw) end); if not ok2 or not d then return end
    for k,v in pairs(d) do if v~=nil then AS[k]=v end end
    local function ci(k,lo,hi,def)
        local v=AS[k]
        if type(v)~="number" then AS[k]=def else AS[k]=math.clamp(math.floor(v),lo,hi) end
    end
    ci("stSpdI",1,#CFG.SPEEDS,CFG.DEF_SPD)
    ci("slowRecI",1,#CFG.SLOW_FACTS,CFG.DEF_SLOW)
    ci("autoCkptEvery",15,600,60)
    ci("stAutoSaveEvery",60,1800,300)
    ci("stCameraFOV",40,120,70)
    if type(AS.cfgSize)~="string" or not CFG_PANEL_SIZES[AS.cfgSize] then AS.cfgSize="medium" end
end
local function saveAS()
    if not hasFile then return end
    pcall(function() makefolder("TAS") end)
    pcall(writefile,SETTINGS_FILE,Http:JSONEncode(AS))
end

local function sCF(c) return {c:GetComponents()} end
local function dCF(t) return CFrame.new(table.unpack(t)) end
local function sV3(v) return {v.X,v.Y,v.Z} end

local function I(cls,pr)
    local o=Instance.new(cls)
    for k,v in pairs(pr) do if k~="Parent" then pcall(function() o[k]=v end) end end
    if pr.Parent then o.Parent=pr.Parent end
    return o
end
local function getChar()

local function deepCopy(t)
    if type(t)~="table" then return t end
    local c={}
    for k,v in pairs(t) do c[deepCopy(k)]=deepCopy(v) end
    return c
end
local function hasValue(arr,val)
    for _,v in ipairs(arr or {}) do if v==val then return true end end
    return false
end

local function markDirty(v) local s=sl(); if s then s.dirty=(v~=false) end end

    if not s.lastSavedHash then

    s.dirty=(slotHash(s)~=s.lastSavedHash)

local function pushUndo()
    local s=sl(); if not s then return end
    if not s.undoStack then s.undoStack={} end
    if not s.redoStack then s.redoStack={} end
    s.undoStack[#s.undoStack+1]={
        frames=deepCopy(s.frames),
        checkpoints=deepCopy(s.checkpoints or {}),
        checkpointNames=deepCopy(s.checkpointNames or {}),
        frame=S.frame
    }
    while #s.undoStack>MAX_UNDO do table.remove(s.undoStack,1) end
    s.redoStack={}
end

local function nearestCheckpointIndex(frameIdx)
    local s=sl(); if not s or not s.checkpoints or #s.checkpoints==0 then return nil end
    local best=nil; local bestDist=math.huge
    for _,fi in ipairs(s.checkpoints) do
        local d=math.abs(fi-frameIdx)
        if d<bestDist then bestDist=d; best=fi end
    end
    return best
end

local function maybeMilestoneNotify()
    if not S.notifyFrameMilestones then return end
    local f=S.frame
    if f>=60 and f%60==0 and f~=S.lastMilestone then
        S.lastMilestone=f
        notify("Frame "..f.." / "..("%.2f"):format(f*DT).."s",P.auto)
    end
end

---------- SLOW GRAVITY ----------
local function removeSlowGravity()
    if S.slowVF then pcall(function() S.slowVF:Destroy() end); S.slowVF=nil end
    if S.slowAtt then pcall(function() S.slowAtt:Destroy() end); S.slowAtt=nil end
end
local function applySlowGravity()
    removeSlowGravity()
    local c,h=getChar(); if not c or not h then return end
    local factor=sf(); if factor>=1 then return end
    local mass=0
    for _,p in ipairs(c:GetDescendants()) do
        if p:IsA("BasePart") then mass=mass+p:GetMass() end
    end
    if mass<=0 then return end
    local att=Instance.new("Attachment"); att.Name="TAS_SGAtt"; att.Parent=h
    local vf=Instance.new("VectorForce"); vf.Name="TAS_SGForce"; vf.Attachment0=att
    vf.RelativeTo=Enum.ActuatorRelativeTo.World
    vf.Force=Vector3.new(0,mass*workspace.Gravity*(1-factor),0)
    vf.Parent=h
    S.slowAtt=att
    S.slowVF=vf
end
local function refreshSlowGravity()
    if S.mode~="recording" and S.mode~="rec_paused" then return end
    if not S.slowRec then removeSlowGravity() else applySlowGravity() end
end

---------- ANIMATIONS ----------
local function capAnims()
    local _,_,_,an=getChar(); if not an then return {} end
    local r={}
    local factor=(S.mode=="recording" and S.slowRec) and sf() or 1
    pcall(function()
        for _,t in ipairs(an:GetPlayingAnimationTracks()) do
            if t.Animation then
                r[#r+1]={
                    id=t.Animation.AnimationId,
                    sp=(factor>0) and (t.Speed/factor) or t.Speed,
                    wt=t.WeightCurrent,
                    tp=t.TimePosition
                }
            end
        end
    end)
    return r
end
local function clearAC()
    for _,t in pairs(S.aC) do pcall(function() t:Stop(0); t:Destroy() end) end
    S.aC={}
end
local function stopPAnims()
    local _,_,_,an=getChar(); if not an then return end
    pcall(function() for _,t in ipairs(an:GetPlayingAnimationTracks()) do t:Stop(0) end end)
end
local function getTrack(an,id)
    local t=S.aC[id]
    if t then
        local ok,v=pcall(function() return t.Animation end)
        if ok and v then return t end
        S.aC[id]=nil
    end
    local a=Instance.new("Animation"); a.AnimationId=id
    local ok,tr=pcall(function() return an:LoadAnimation(a) end)
    if ok and tr then
        tr.Priority=Enum.AnimationPriority.Action4
        S.aC[id]=tr
        return tr
    end
end
local function applyAnims(anims,paused)
    if not S.anim then return end
    local _,_,_,an=getChar(); if not an then return end
    local need={}
    local ps=paused and 0 or spd()
    for _,a in ipairs(anims or {}) do
        need[a.id]=true
        local t=getTrack(an,a.id)
        if t then
            pcall(function()
                if not t.IsPlaying then t:Play(0,a.wt,0) end
                t:AdjustSpeed(a.sp*ps)
                t:AdjustWeight(a.wt,0)
                t.TimePosition=a.tp
            end)
        end
    end
    for id,t in pairs(S.aC) do
        if not need[id] then pcall(function() if t.IsPlaying then t:Stop(0) end end) end
    end
end

local function restoreCharacter()
    local _,h,hu=getChar()
    if hu then hu.WalkSpeed=S.sWS; hu.JumpPower=S.sJP end
    if h then
        h.AssemblyLinearVelocity=Vector3.zero

    task.delay(0.05,function()

        local _,_,hu2=getChar()
        if hu2 then pcall(function() hu2:ChangeState(Enum.HumanoidStateType.Running) end) end
    end)
end

---------- FRAME CAPTURE / APPLY ----------
local function capFrame()
    local c,h,hu=getChar(); if not c or not h or not hu then return end
    local f={
        cf=sCF(h.CFrame),
        v=sV3(h.AssemblyLinearVelocity),
        ws=hu.WalkSpeed,
        jp=hu.JumpPower,
        st=hu:GetState().Value,
        an=capAnims()
    }
    local tl=c:FindFirstChildOfClass("Tool")
    if tl then f.tl=tl.Name end
    if S.camR and not (S.slowRec and not S.slowRecCam) then f.cm=sCF(cam.CFrame) end
    return f
end

local function applyFrame(idx,paused)
    local s=sl(); if not s or not s.frames[idx] then return end
    local f=s.frames[idx]
    local c,h,hu=getChar(); if not c or not h or not hu then return end

    if S.playbackFrameBlend and idx>1 and s.frames[idx-1] and not paused then
        local prev=dCF(s.frames[idx-1].cf)
        local cur=dCF(f.cf)
        h.CFrame=prev:Lerp(cur,0.5)
    else
        h.CFrame=dCF(f.cf)
    end

    h.AssemblyLinearVelocity=Vector3.zero
    h.AssemblyAngularVelocity=Vector3.zero

    if S.holdPose then
        hu.WalkSpeed=0; hu.JumpPower=0
        pcall(function() hu:ChangeState(Enum.HumanoidStateType.Physics) end)
    else
        hu.WalkSpeed=f.ws or 0
        hu.JumpPower=f.jp or 0
    end

    applyAnims(f.an,paused)
    if S.camR and f.cm then
        cam.CameraType=Enum.CameraType.Scriptable
        cam.CFrame=dCF(f.cm)
    end
end

local function addCheckpointAtFrame(fi,quiet)
    local s=sl(); if not s or fi<1 or fi>#s.frames then return false end
    s.checkpoints=s.checkpoints or {}
    s.checkpointNames=s.checkpointNames or {}
    if hasValue(s.checkpoints,fi) then return false end
    s.checkpoints[#s.checkpoints+1]=fi
    table.sort(s.checkpoints)
    if S.autoNameCheckpoints and not s.checkpointNames[tostring(fi)] then
        s.checkpointNames[tostring(fi)]="F"..fi
    end
    markDirty(true)
    if S.trail then refreshCkptMarkers() end
    if not quiet then notify("Checkpoint F"..fi,P.ckpt) end
    return true
end

local function updateSeekMarker()
    local s=sl()
    if not S.trail or not s or #s.frames==0 or S.frame<1 or not s.frames[S.frame] then
        clearSeekMarker()
        return
    end
    ensureSeekMarker()
    local pos=dCF(s.frames[S.frame].cf).Position
    S.seekMarker.Position=pos+Vector3.new(0,0.15,0)
    S.seekLine.CFrame=CFrame.new(pos+Vector3.new(0,1.3,0))
    if S.seekText then
        local isCk=hasValue(s.checkpoints or {},S.frame)
        local tag=""
        if isCk then
            local idx=1
            for i,fi in ipairs(s.checkpoints or {}) do if fi==S.frame then idx=i break end end
            tag="★ "..checkpointLabel(S.frame,idx).." "
        end
        S.seekText.Text=tag.."F"..tostring(S.frame)
    end
end

        local f=s.frames[i]
        if f and f.cm then
            local cf=dCF(f.cm)
            local pos=cf.Position
            if not last or (pos-last).Magnitude>=2.5 then
                I("Part",{
                    Size=Vector3.new(0.22,0.22,0.22),Shape=Enum.PartType.Ball,Position=pos,
                    Anchored=true,CanCollide=false,CanTouch=false,CanQuery=false,Material=Enum.Material.Neon,
                    Color=P.camGuide,Transparency=0.2,CastShadow=false,Parent=fo
                })
                if last then
                    local dist=(pos-last).Magnitude
                    if dist>0.05 then
                        local mid=(pos+last)*0.5
                        I("Part",{
                            Size=Vector3.new(0.08,0.08,dist),CFrame=CFrame.new(mid,pos),
                            Anchored=true,CanCollide=false,CanTouch=false,CanQuery=false,Material=Enum.Material.Neon,
                            Color=P.camGuide,Transparency=0.55,CastShadow=false,Parent=fo
                        })
                    end
                end
                last=pos
            end
        end
    end
end

---------- RECORDING ----------
local function startRec()
    if not getChar() then notify("No character",P.rec); return end
    local s=sl()
    pushUndo()
    local isContinue=(#s.frames>0) and (S.frame>0) and (S.frame<#s.frames)
    if isContinue then
        for i=#s.frames,S.frame+1,-1 do table.remove(s.frames,i) end
        if s.checkpoints then
            for i=#s.checkpoints,1,-1 do
                if s.checkpoints[i]>=S.frame then
                    s.checkpointNames[tostring(s.checkpoints[i])]=nil
                    table.remove(s.checkpoints,i)
                end
            end
            s.checkpoints[#s.checkpoints+1]=S.frame
            if S.autoNameCheckpoints then s.checkpointNames[tostring(S.frame)]="Branch" end
            refreshCkptMarkers()
        end
        notify("Rec continue from F"..S.frame,P.ckpt)
        markDirty(true)
    elseif S.frame<=0 then
        s.frames={}
        s.checkpoints={}
        s.checkpointNames={}
        clearCkptParts()
        markDirty(true)
    end
    S.mode="recording"
    S.rAcc=0
    S.autoSaveAcc=0
    clearAC()
    S.lastMilestone=0

    local _,h,hu=getChar()
    if h then
        h.AssemblyLinearVelocity=Vector3.zero
        h.AssemblyAngularVelocity=Vector3.zero
    end
    if hu then
        local factor=S.slowRec and sf() or 1
        hu.WalkSpeed=S.sWS*factor
        hu.JumpPower=S.sJP*factor
        pcall(function() hu:ChangeState(Enum.HumanoidStateType.GettingUp) end)
        task.defer(function()
            if S.mode=="recording" then pcall(function() hu:ChangeState(Enum.HumanoidStateType.Running) end) end
        end)
    end
    if S.camR then cam.CameraType=Enum.CameraType.Custom end
    if S.slowRec then applySlowGravity() end
    if S.freezeLighting then captureLighting() end
    notify("REC"..(S.slowRec and (" Slow "..tostring(sf()).."x") or ""),P.rec)
end

local function recPause()
    if S.mode~="recording" then return end
    local _,h,hu,an=getChar()
    if h then
        h.AssemblyLinearVelocity=Vector3.zero
        h.AssemblyAngularVelocity=Vector3.zero
    end
    if hu then
        hu.WalkSpeed=0; hu.JumpPower=0
        pcall(function() hu:ChangeState(Enum.HumanoidStateType.Physics) end)
    end
    if an then pcall(function() for _,t in ipairs(an:GetPlayingAnimationTracks()) do t:AdjustSpeed(0) end end) end
    S.mode="rec_paused"
    notify("REC Paused F"..S.frame,P.pause)
end

local function recResume()
    if S.mode~="rec_paused" then return end
    local _,_,hu,an=getChar()
    local factor=S.slowRec and sf() or 1
    if hu then
        hu.WalkSpeed=S.sWS*factor
        hu.JumpPower=S.sJP*factor
        pcall(function() hu:ChangeState(Enum.HumanoidStateType.Running) end)
    end
    if an then pcall(function() for _,t in ipairs(an:GetPlayingAnimationTracks()) do t:AdjustSpeed(factor) end end) end
    S.mode="recording"
    notify("REC Resumed F"..S.frame,P.rec)
end

local function stopRec()
    removeSlowGravity()
    local _,_,hu,an=getChar()
    if hu then hu.WalkSpeed=S.sWS; hu.JumpPower=S.sJP end
    if an and S.slowRec then pcall(function() for _,t in ipairs(an:GetPlayingAnimationTracks()) do t:AdjustSpeed(1) end end) end
    S.mode="idle"
    S.frame=fc()
    refreshDirty()
    updateSeekMarker()
    if S.autoTrailRefresh and S.trail then
        showTrail()
        refreshCamGuide()
        refreshVelGuide()
    end
    notify("Stop "..fc().."F ("..("%.1f"):format(fc()*DT).."s)",P.play)
end

local function recTick(dt)
    if S.mode~="recording" then return end
    local factor=S.slowRec and sf() or 1
    if S.slowRec then
        local _,_,_,an=getChar()
        if an then pcall(function() for _,t in ipairs(an:GetPlayingAnimationTracks()) do t:AdjustSpeed(factor) end end) end
    end
    S.rAcc=S.rAcc+dt*factor
    S.autoSaveAcc=S.autoSaveAcc+dt
    while S.rAcc>=DT do
        S.rAcc=S.rAcc-DT
        local f=capFrame()
        if f then
            table.insert(sl().frames,f)
            S.frame=#sl().frames
            markDirty(true)
            maybeMilestoneNotify()

            if S.autoCkpt and S.autoCkptEvery>0 and S.frame>0 and (S.frame % S.autoCkptEvery==0) then
                if addCheckpointAtFrame(S.frame,true) then
                    if sl().checkpointNames then sl().checkpointNames[tostring(S.frame)]="A"..S.frame end
                    notify("Auto checkpoint F"..S.frame,P.auto)
                end
            end
        end
    end
    if S.autoSave and hasFile and S.autoSaveAcc>=S.autoSaveEvery then
        S.autoSaveAcc=0
        saveSlotDirect(S.slot)
        notify("AutoSaved",P.play)
    end
end

    local _,h,hu=getChar(); if not h then return end
    h.AssemblyLinearVelocity=Vector3.zero
    h.AssemblyAngularVelocity=Vector3.zero
    if hu then
        hu.WalkSpeed=0; hu.JumpPower=0
        pcall(function() hu:ChangeState(Enum.HumanoidStateType.Physics) end)
    end
end

---------- PLAYBACK ----------
local function startPlay()
    local s=sl(); if not s or #s.frames==0 then notify("No frames",P.rec); return end
    local c,h,hu=getChar(); if not c or not h or not hu then notify("No character",P.rec); return end
    if hu.WalkSpeed>0 then S.sWS=hu.WalkSpeed end
    if hu.JumpPower>0 then S.sJP=hu.JumpPower end
    if S.frame>=#s.frames or S.frame<1 then S.frame=1 end
    S.mode="playing"
    S.acc=0
    stopPAnims()
    clearAC()
    hu.WalkSpeed=0; hu.JumpPower=0
    if S.freezeLighting then captureLighting() end
    updateSeekMarker()
    notify("Play F"..S.frame,P.play)
end

local function pausePlay()
    if S.mode=="playing" then
        S.mode="paused"
        local s=sl(); if s and s.frames[S.frame] then applyAnims(s.frames[S.frame].an,true) end
        local _,h,hu=getChar()
        if h then
            h.AssemblyLinearVelocity=Vector3.zero
            h.AssemblyAngularVelocity=Vector3.zero
        end
        if hu then
            hu.WalkSpeed=0; hu.JumpPower=0
            pcall(function() hu:ChangeState(Enum.HumanoidStateType.Physics) end)
        end
        updateSeekMarker()
        notify("Paused F"..S.frame,P.pause)
    elseif S.mode=="paused" then
        S.mode="playing"
        S.acc=0
        updateSeekMarker()
        notify("Resume",P.play)
    end
end

local function stopPlay()
    if S.addFramePending then
        S.addFramePending=false; S.addFrameAcc=0
        local _,h,hu=getChar()
        if h then
            h.AssemblyLinearVelocity=Vector3.zero
            h.AssemblyAngularVelocity=Vector3.zero
        end
        if hu then hu.WalkSpeed=0; hu.JumpPower=0 end
    end
    S.mode="idle"
    clearAC()
    stopPAnims()
    restoreCharacter()
    updateSeekMarker()
    notify("Stop",P.dim)
end

local function stopAll()
    if S.mode=="recording" or S.mode=="rec_paused" then
        stopRec()
    elseif S.mode~="idle" then
        stopPlay()
    end
end

local function seekTo(idx)
    local s=sl(); if not s or #s.frames==0 then return end
    local target=math.clamp(idx,1,#s.frames)
    if S.snapToCkpt and s.checkpoints and #s.checkpoints>0 then
        local near=nearestCheckpointIndex(target)
        if near and math.abs(near-target)<=2 then target=near end
    end
    S.frame=target
    if S.mode=="playing" or S.mode=="paused" then
        applyFrame(S.frame,S.mode~="playing")
    end
    updateSeekMarker()
end

local function playTick(dt)
    if S.mode~="playing" then return end
    local s=sl(); if not s or #s.frames==0 then stopPlay(); return end
    S.acc=S.acc+dt*spd()
    while S.acc>=DT do
        S.acc=S.acc-DT
        S.frame=S.frame+1
        if S.frame>#s.frames then
            if S.loop then
                S.frame=1; S.acc=0
            else
                S.frame=#s.frames
                pausePlay()
                return
            end
        end
        if S.stopAtCheckpoint and hasValue(s.checkpoints or {},S.frame) and S.frame<#s.frames then
            applyFrame(S.frame,true)
            S.mode="paused"
            notify("Auto pause checkpoint F"..S.frame,P.ckpt)
            updateSeekMarker()
            return
        end
    end
    applyFrame(S.frame,false)
    updateSeekMarker()
end

local function holdTick()
    if S.addFramePending then return end
    if S.mode~="paused" then return end
    if not S.holdPose then return end
    local s=sl(); if not s or not s.frames[S.frame] then return end
    local f=s.frames[S.frame]
    local _,h,hu=getChar(); if not h then return end
    h.CFrame=dCF(f.cf)
    h.AssemblyLinearVelocity=Vector3.zero
    h.AssemblyAngularVelocity=Vector3.zero
    pcall(function() hu:ChangeState(Enum.HumanoidStateType.Physics) end)
    if hu then hu.WalkSpeed=0; hu.JumpPower=0 end
end

---------- TRIM ----------
local function trimHere()
    local s=sl()
    if not s or #s.frames==0 or S.frame<1 or S.frame>=#s.frames then
        notify("Can't trim",P.rec); return
    end
    pushUndo()
    local rm=#s.frames-S.frame
    for i=#s.frames,S.frame+1,-1 do table.remove(s.frames,i) end
    if s.checkpoints then
        for i=#s.checkpoints,1,-1 do
            if s.checkpoints[i]>S.frame then
                s.checkpointNames[tostring(s.checkpoints[i])]=nil
                table.remove(s.checkpoints,i)
            end
        end
        refreshCkptMarkers()
    end
    markDirty(true)
    if S.trail then showTrail() else updateSeekMarker() end
    notify("Trim end -"..rm.." -> "..#s.frames.."F",P.pause)
end
local function trimFront()
    local s=sl()
    if not s or #s.frames==0 or S.frame<=1 then
        notify("Can't trim",P.rec); return
    end
    pushUndo()
    local rm=S.frame-1
    for _=1,rm do table.remove(s.frames,1) end
    if s.checkpoints then
        local nc={}
        local nn={}
        for _,fi in ipairs(s.checkpoints) do
            local n=fi-rm
            if n>=1 then
                nc[#nc+1]=n
                if s.checkpointNames[tostring(fi)] then nn[tostring(n)]=s.checkpointNames[tostring(fi)] end
            end
        end
        s.checkpoints=nc
        s.checkpointNames=nn
        refreshCkptMarkers()
    end
    S.frame=1
    markDirty(true)
    if S.trail then showTrail() else updateSeekMarker() end
    notify("Trim front -"..rm.." -> "..#s.frames.."F",P.pause)
end

---------- GHOST ----------
local function killGhost()
    if S.gPart then pcall(function() S.gPart:Destroy() end); S.gPart=nil end
    if S.gClone then pcall(function() S.gClone.model:Destroy() end); S.gClone=nil end
    S.gBuildType=nil
end
local function buildEmojiGhost()
    if S.gPart then return end
    S.gPart=I("Part",{
        Name="TAS_Ghost",Size=Vector3.new(2,5,2),Anchored=true,CanCollide=false,CanTouch=false,CanQuery=false,
        Material=Enum.Material.ForceField,BrickColor=BrickColor.new("Electric blue"),Transparency=0.4,
        CastShadow=false,Parent=workspace
    })
    local bb=I("BillboardGui",{Size=UDim2.new(0,36,0,36),StudsOffset=Vector3.new(0,3.5,0),AlwaysOnTop=true,Parent=S.gPart})
    I("TextLabel",{Size=UDim2.new(1,0,1,0),BackgroundTransparency=1,Text="👻",TextScaled=true,TextColor3=Color3.new(1,1,1),Parent=bb})
end
local function buildCloneGhost()
    if S.gClone then return end
    local c=plr.Character; if not c then return end
    local clone=c:Clone(); clone.Name="TAS_GhostClone"
    for _,v in ipairs(clone:GetDescendants()) do
        if v:IsA("Script") or v:IsA("LocalScript") or v:IsA("ModuleScript") or v:IsA("Animator") then v:Destroy() end
    end
    local hum=clone:FindFirstChildOfClass("Humanoid"); if hum then hum:Destroy() end
    local cHRP=clone:FindFirstChild("HumanoidRootPart")
    local parts={}
    if cHRP then
        local inv=cHRP.CFrame:Inverse()
        for _,v in ipairs(clone:GetDescendants()) do
            if v:IsA("BasePart") then
                parts[v]=inv*v.CFrame
                v.Anchored=true; v.CanCollide=false; v.CanTouch=false; v.CanQuery=false; v.CastShadow=false
                v.Transparency=(v==cHRP) and 1 or math.clamp(v.Transparency+0.45,0,0.9)
                v.LocalTransparencyModifier=0
            end
        end
    end
    clone.Parent=workspace
    S.gClone={model=clone,hrp=cHRP,parts=parts}
end
local function ghostTick(dt)
    if not S.ghost then killGhost(); return end
    local s=sl(); if not s or #s.frames==0 then return end
    if S.ghostType~=S.gBuildType then
        killGhost()
        S.gF=1
        S.gA=0
        S.gBuildType=S.ghostType
    end
    if S.ghostType=="clone" then
        if S.gPart then pcall(function() S.gPart:Destroy() end); S.gPart=nil end
        buildCloneGhost()
    else
        if S.gClone then pcall(function() S.gClone.model:Destroy() end); S.gClone=nil end
        buildEmojiGhost()
    end
    S.gA=S.gA+dt
    while S.gA>=DT do
        S.gA=S.gA-DT
        S.gF=S.gF+1
        if S.gF>#s.frames then S.gF=1 end
    end
    local f=s.frames[S.gF]; if not f then return end
    local cf=dCF(f.cf)
    if S.ghostType=="clone" and S.gClone and S.gClone.hrp then
        S.gClone.hrp.CFrame=cf
        for part,offset in pairs(S.gClone.parts) do
            pcall(function() if part.Parent then part.CFrame=cf*offset end end)
        end
    elseif S.gPart then
        S.gPart.CFrame=cf
    end
end

---------- FILE I/O ----------
function saveSlotDirect(idx)
    if not hasFile then notify("No file API",P.pause); return end
    pcall(function() makefolder("TAS") end)
    pcall(function() makefolder(PLACE_DIR) end)
    local s=S.slots[idx]; if not s then return end
    s.meta.ts=os.time()
    local ok=pcall(writefile,CFG.DIR.."slot_"..idx..".json",Http:JSONEncode({
        n=s.name,m=s.meta,f=s.frames,c=s.checkpoints or{},cn=s.checkpointNames or {}
    }))
    if ok then
        local mf={}
        for i,v in pairs(S.slots) do
            mf[tostring(i)]={n=v.name,fc=#v.frames,dirty=v.dirty or false}
        end
        pcall(writefile,CFG.MAN,Http:JSONEncode(mf))
        s.lastSavedHash=slotHash(s)
        s.dirty=false
        notify("Saved #"..idx.." ["..mapName.."]",P.play)
    else
        notify("Save failed",P.rec)
    end
end
local function loadSlot(idx)
    if not hasFile then return false end
    local ok,raw=pcall(readfile,CFG.DIR.."slot_"..idx..".json")
    if not ok or not raw then return false end
    local ok2,d=pcall(function() return Http:JSONDecode(raw) end)
    if not ok2 or not d then return false end
    local s={
        name=d.n or ("Slot "..idx),
        frames=d.f or {},
        checkpoints=d.c or {},
        checkpointNames=d.cn or {},
        meta=d.m or {},
        dirty=false,lastSavedHash=nil,undoStack={},redoStack={}
    }
    s.lastSavedHash=Http:JSONEncode({
        n=s.name,m=s.meta,f=s.frames,c=s.checkpoints or {},cn=s.checkpointNames or {}
    })
    S.slots[idx]=s
    return true
end
local function loadAll()
    if not hasFile then return end
    local ok,raw=pcall(readfile,CFG.MAN); if not ok or not raw then return end
    local ok2,mf=pcall(function() return Http:JSONDecode(raw) end); if not ok2 or not mf then return end
    for k in pairs(mf) do
        local i=tonumber(k)
        if i then loadSlot(i) end
    end
end

---------- UNDO / REDO ----------
local function applyUndoState(state)
    if S.mode~="idle" then stopAll() end
    local s=sl(); if not s then return end
    s.frames=deepCopy(state.frames)
    s.checkpoints=deepCopy(state.checkpoints or {})
    s.checkpointNames=deepCopy(state.checkpointNames or {})
    S.frame=math.clamp(state.frame or S.frame,0,#s.frames)
    if S.frame<1 and #s.frames>0 then S.frame=1 end
    markDirty(true)
end
local function undo()
    local s=sl(); if not s or not s.undoStack or #s.undoStack==0 then notify("Nothing to undo",P.rec); return end
    s.redoStack=s.redoStack or {}
    s.redoStack[#s.redoStack+1]={
        frames=deepCopy(s.frames),
        checkpoints=deepCopy(s.checkpoints or {}),
        checkpointNames=deepCopy(s.checkpointNames or {}),
        frame=S.frame
    }
    applyUndoState(table.remove(s.undoStack))
    if S.trail then showTrail() else clearTrail(); clearCkptParts(); refreshCkptMarkers() end
    updateSeekMarker()
    notify("Undo ("..#s.undoStack.." left)",P.pause)
end
local function redo()
    local s=sl(); if not s or not s.redoStack or #s.redoStack==0 then notify("Nothing to redo",P.rec); return end
    s.undoStack=s.undoStack or {}
    s.undoStack[#s.undoStack+1]={
        frames=deepCopy(s.frames),
        checkpoints=deepCopy(s.checkpoints or {}),
        checkpointNames=deepCopy(s.checkpointNames or {}),
        frame=S.frame
    }
    applyUndoState(table.remove(s.redoStack))
    if S.trail then showTrail() else clearTrail(); clearCkptParts(); refreshCkptMarkers() end
    updateSeekMarker()
    notify("Redo ("..#s.redoStack.." left)",P.acc)
end

---------- ADD ONE FRAME ----------
local function addOneFrame()
    if S.mode~="paused" then return end
    if S.frame~=fc() or fc()==0 then return end
    if S.addFramePending then return end
    pushUndo()
    S.addFramePending=true
    S.addFrameAcc=0
    local _,h,hu=getChar()
    if hu then
        hu.WalkSpeed=S.sWS; hu.JumpPower=S.sJP
        pcall(function() hu:ChangeState(Enum.HumanoidStateType.Running) end)
    end
    if h then
        h.AssemblyLinearVelocity=Vector3.zero
        h.AssemblyAngularVelocity=Vector3.zero
    end
    notify("+1F... Move slightly",P.acc)
end

local function addFrameTick(dt)
    if not S.addFramePending then return end
    S.addFrameAcc=S.addFrameAcc+dt
    if S.addFrameAcc>=DT then
        S.addFramePending=false; S.addFrameAcc=0
        local f=capFrame(); local s=sl()
        if f and s then
            table.insert(s.frames,f)
            S.frame=#s.frames
            markDirty(true)
            if S.autoCkpt and S.autoCkptEvery>0 and S.frame % S.autoCkptEvery==0 then
                addCheckpointAtFrame(S.frame,true)
            end
            if S.trail then showTrail() else updateSeekMarker() end
        end
        local _,h,hu=getChar()
        if h then
            h.AssemblyLinearVelocity=Vector3.zero
            h.AssemblyAngularVelocity=Vector3.zero
        end
        if hu then
            hu.WalkSpeed=0; hu.JumpPower=0
            pcall(function() hu:ChangeState(Enum.HumanoidStateType.Physics) end)
        end
        notify("+1F -> "..fc().."F",P.acc)
    end
end

-- ─── Wire Events ─────────────────────────
local function wireEvents()
    local td=false; local ti=nil; local ts=nil; local tp=nil; local tdid=false
    uiTogBtn.InputBegan:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
            td=true; ti=inp; tdid=false; ts=inp.Position; tp=uiTogBtn.Position
        end
    end)
    uiTogBtn.InputEnded:Connect(function(inp)
        if td and inp==ti then
            td=false; ti=nil
            if tdid then
                AS.togBtnXS=uiTogBtn.Position.X.Scale; AS.togBtnXO=uiTogBtn.Position.X.Offset
                AS.togBtnYS=uiTogBtn.Position.Y.Scale; AS.togBtnYO=uiTogBtn.Position.Y.Offset
                saveAS()
            end
        end
    end)
    UIS.InputChanged:Connect(function(inp)
        if td and inp==ti then
            local d=inp.Position-ts
            if d.Magnitude>6 then tdid=true end
            if tdid then uiTogBtn.Position=UDim2.new(tp.X.Scale,tp.X.Offset+d.X,tp.Y.Scale,tp.Y.Offset+d.Y) end
        end
    end)
    uiTogBtn.MouseButton1Click:Connect(function()
        if not tdid then
            S.uiOpen=not S.uiOpen
            main.Visible=S.uiOpen
        end
    end)

    local slD=false; local slI=nil
    local function slUpd(x)
        local total=fc(); if total==0 then return end
        local rel=math.clamp((x-sliderBg.AbsolutePosition.X)/sliderBg.AbsoluteSize.X,0,1)
        local idx=math.max(1,math.ceil(rel*total))
        seekTo(idx)
        if S.mode=="playing" then
            S.mode="paused"
            applyFrame(S.frame,true)
        elseif S.mode=="paused" then
            applyFrame(S.frame,true)
        end
        updateSeekMarker()
    end
    sliderTouch.InputBegan:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
            slD=true; slI=inp; slUpd(inp.Position.X)
        end
    end)
    sliderTouch.InputEnded:Connect(function(inp) if inp==slI then slD=false; slI=nil; updateSeekMarker() end end)
    UIS.InputChanged:Connect(function(inp) if slD and inp==slI then slUpd(inp.Position.X) end end)

    cfgBtn.MouseButton1Click:Connect(function()
        S.cfgOpen=not S.cfgOpen
        cfgPanel.Visible=S.cfgOpen
        cfgBtn.BackgroundColor3=S.cfgOpen and P.togOn or P.btn
    end)
    cfgCloseBtn.MouseButton1Click:Connect(function()
        S.cfgOpen=false
        cfgPanel.Visible=false
        cfgBtn.BackgroundColor3=P.btn
    end)

    recBtn.MouseButton1Click:Connect(function()
        if S.mode=="idle" then
            startRec()
        elseif S.mode=="recording" then
            stopRec()
        elseif S.mode=="rec_paused" then
            recResume()
        elseif (S.mode=="paused" or S.mode=="playing") and S.recContinue then
            local wf=fc()
            stopPlay()
            S.frame=wf
            startRec()
        end
    end)

    pauseBtn.MouseButton1Click:Connect(function()
        if S.mode=="recording" then
            recPause()
        elseif S.mode=="rec_paused" then
            recResume()
        elseif S.mode=="playing" or S.mode=="paused" then
            pausePlay()
        end
    end)

    playBtn.MouseButton1Click:Connect(function()
        if S.mode=="idle" then
            startPlay()
        elseif S.mode=="paused" then
            pausePlay()
        end
    end)

    stopBtn.MouseButton1Click:Connect(function()
        if S.mode=="recording" or S.mode=="rec_paused" then stopRec() elseif S.mode~="idle" then stopPlay() end
    end)

    local function stepBy(n)
        if fc()==0 then return end
        if S.mode=="idle" then
            seekTo(S.frame+n)
            local s=sl()
            if s and s.frames[S.frame] then
                local _,h=getChar()
                if h then h.CFrame=dCF(s.frames[S.frame].cf) end
            end
            restoreCharacter()
            updateSeekMarker()
        elseif S.mode=="paused" then
            seekTo(S.frame+n)
        end
    end

    s1L.MouseButton1Click:Connect(function() stepBy(-1) end)
    s1R.MouseButton1Click:Connect(function() stepBy(1) end)
    s10L.MouseButton1Click:Connect(function() stepBy(-10) end)
    s10R.MouseButton1Click:Connect(function() stepBy(10) end)

    spdDown.MouseButton1Click:Connect(function() S.spdI=math.max(1,S.spdI-1); saveFeatStates() end)
    spdUp.MouseButton1Click:Connect(function() S.spdI=math.min(#CFG.SPEEDS,S.spdI+1); saveFeatStates() end)

    trimFrontBtn.MouseButton1Click:Connect(trimFront)
    trimBtn.MouseButton1Click:Connect(trimHere)

    ghostBtn.MouseButton1Click:Connect(function()
        S.ghost=not S.ghost
        ghostBtn.BackgroundColor3=S.ghost and P.togOn or P.btn
        if not S.ghost then killGhost() end
        saveFeatStates()
        notify(S.ghost and "Ghost ON" or "Ghost OFF",P.acc)
    end)
    loopBtn.MouseButton1Click:Connect(function()
        S.loop=not S.loop
        loopBtn.BackgroundColor3=S.loop and P.togOn or P.btn
        saveFeatStates()
        notify(S.loop and "Loop ON" or "Loop OFF",P.acc)
    end)
    trailBtn.MouseButton1Click:Connect(function()
        S.trail=not S.trail
        trailBtn.BackgroundColor3=S.trail and P.togOn or P.btn
        if S.trail then showTrail() else clearTrail(); clearCkptParts() end
        saveFeatStates()
        notify(S.trail and "Trail ON" or "Trail OFF",P.acc)
    end)

    delBtn.MouseButton1Click:Connect(function()
        if S.mode~="idle" then stopAll() end
        local s=sl()
        if #s.frames>0 or #(s.checkpoints or {})>0 then pushUndo() end
        s.frames={}
        s.checkpoints={}
        s.checkpointNames={}
        S.frame=0
        markDirty(true)
        clearCkptParts()
        clearTrail()
        S.trail=false
        trailBtn.BackgroundColor3=P.btn
        _tickKey=""
        notify("Cleared",P.pause)
    end)

    saveBtn.MouseButton1Click:Connect(function()
        local s=sl(); if not s then return end
        refreshDirty()
        if s.dirty then openSaveConfirm() else saveSlotDirect(S.slot) end
    end)
    confirmYes.MouseButton1Click:Connect(function() closeSaveConfirm(); saveSlotDirect(S.slot); refreshTabs() end)
    confirmNo.MouseButton1Click:Connect(function() closeSaveConfirm(); notify("Save cancelled",P.pause) end)

    loadBtn.MouseButton1Click:Connect(function()
        local s=sl(); refreshDirty()
        if s and s.dirty then openLoadConfirm() else doLoadSlot() end
    end)
    loadConfirmYes.MouseButton1Click:Connect(function() closeLoadConfirm(); doLoadSlot() end)
    loadConfirmNo.MouseButton1Click:Connect(function() closeLoadConfirm(); notify("Load cancelled",P.pause) end)

    undoBtn.MouseButton1Click:Connect(undo)
    redoBtn.MouseButton1Click:Connect(redo)
    addFrameBtn.MouseButton1Click:Connect(addOneFrame)

    UIS.InputBegan:Connect(function(inp,gpe)
        if gpe then return end
        if inp.UserInputType~=Enum.UserInputType.Keyboard then return end
        local k=inp.KeyCode
        local ctrl=UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.RightControl)
        local shift=UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.RightShift)

        if k==Enum.KeyCode.Z and ctrl and shift then redo(); return
        elseif k==Enum.KeyCode.Z and ctrl then undo(); return
        elseif k==Enum.KeyCode.Y and ctrl and not S.saveConfirmOpen and not S.loadConfirmOpen then redo(); return end

        if S.saveConfirmOpen then
            if k==Enum.KeyCode.Y then closeSaveConfirm(); saveSlotDirect(S.slot); refreshTabs()
            elseif k==Enum.KeyCode.N then closeSaveConfirm(); notify("Save cancelled",P.pause) end
            return
        end
        if S.loadConfirmOpen then
            if k==Enum.KeyCode.Y then closeLoadConfirm(); doLoadSlot()
            elseif k==Enum.KeyCode.N then closeLoadConfirm(); notify("Load cancelled",P.pause) end
            return
        end

        if k==Enum.KeyCode.R then
            if S.mode=="idle" then startRec()
            elseif S.mode=="recording" then stopRec()
            elseif S.mode=="rec_paused" then recResume()
            elseif (S.mode=="paused" or S.mode=="playing") and S.recContinue then local wf=fc(); stopPlay(); S.frame=wf; startRec() end
        elseif k==Enum.KeyCode.P then
            if S.mode=="recording" then recPause()
            elseif S.mode=="rec_paused" then recResume()
            elseif S.mode=="idle" then startPlay()
            elseif S.mode=="playing" or S.mode=="paused" then pausePlay() end
        elseif k==Enum.KeyCode.X then
            if S.mode=="recording" or S.mode=="rec_paused" then stopRec() elseif S.mode~="idle" then stopPlay() end
        elseif k==Enum.KeyCode.Left then
            stepBy(shift and -10 or -1)
        elseif k==Enum.KeyCode.Right then
            stepBy(shift and 10 or 1)
        elseif k==Enum.KeyCode.C and ctrl then
            local s=sl()
            if s and fc()>0 and S.frame>=1 and S.frame<=fc() then
                pushUndo()
                addCheckpointAtFrame(S.frame,false)
                if S.trail then showTrail() else refreshCkptMarkers(); updateSeekMarker() end
            end
        elseif k==Enum.KeyCode.F and ctrl then
            S.focusMode=not S.focusMode
            applyFocusMode(S.focusMode)
            saveFeatStates()
            notify(S.focusMode and "Focus Mode ON" or "Focus Mode OFF",P.focus)
        elseif k==Enum.KeyCode.B and ctrl then
            S.cinematicBars=not S.cinematicBars
            refreshBars()
            saveFeatStates()
            notify(S.cinematicBars and "Cinematic Bars ON" or "Cinematic Bars OFF",P.focus)
        end
    end)
end

local function startLoop()
    local rpt=0
    RunService.RenderStepped:Connect(function(dt)
        recTick(dt)
        playTick(dt)
        holdTick()
        recPausedHold()
        ghostTick(dt)
        addFrameTick(dt)

        if S.freezeLighting then applyFrozenLighting() end
        if S.cameraFOVLock then cam.FieldOfView=S.cameraFOV end

        local total=fc()
        local f=S.frame
        frameLbl.Text="F:"..tostring(f).."/"..tostring(total)
        timeLbl.Text=fmtSec(f*DT).." / "..fmtSec(total*DT)

        local s2=spd()
        spdLbl.Text=tostring(s2).."x"
        spdLbl.TextColor3=s2==1 and P.acc or (s2>1 and P.play or P.pause)

        local _,h= getChar()
        local vel=0
        if h then vel=h.AssemblyLinearVelocity.Magnitude end
        statsLbl.Visible=S.showStats
        statsLbl.Text=("v=%.2f  dt=%.3f  %s"):format(vel,dt,S.mode)
        statsLbl.TextColor3=vel>0.2 and P.safe or P.dim

        if total>0 then
            local pct=math.clamp(f/total,0,1)
            sliderFill.Size=UDim2.new(pct,0,1,0)
            sliderThumb.Position=UDim2.new(0,math.floor(pct*sliderBg.AbsoluteSize.X)-7,0,-5)
        else
            sliderFill.Size=UDim2.new(0,0,1,0)
            sliderThumb.Position=UDim2.new(0,-7,0,-5)
        end

        refreshSliderTicks()
        updateSeekMarker()

        if S.mode=="recording" then
            rpt=rpt+dt
            local a=0.5+0.5*math.sin(rpt*6)
            recBtn.BackgroundColor3=(S.slowRec and P.slow or P.rec):Lerp(Color3.new(0.05,0.05,0.2),a)
            recBtn.Text=S.slowRec and ("Slow"..tostring(f)) or ("REC"..tostring(f))
        elseif S.mode=="rec_paused" then
            rpt=rpt+dt
            local a=0.4+0.4*math.sin(rpt*3)
            recBtn.BackgroundColor3=P.recP:Lerp(Color3.fromRGB(80,50,10),a)
            recBtn.Text="REC||"
        else
            rpt=0
            recBtn.BackgroundColor3=S.slowRec and P.slow or P.rec
            recBtn.Text=S.slowRec and "SlwREC" or "REC"
        end

        pauseBtn.BackgroundColor3=(S.mode=="rec_paused" or S.mode=="paused") and P.pause:Lerp(Color3.new(1,1,1),0.25) or P.pause
        playBtn.BackgroundColor3=S.mode=="playing" and P.play:Lerp(Color3.new(1,1,1),0.15) or P.play

        for i,b in ipairs(tabButtons) do
            local sv=S.slots[i]
            if sv then
                b.BackgroundColor3=i==S.slot and P.tabA or P.btn
                local w=sv.name..(sv.dirty and "*" or "")
                if b.Text~=w then b.Text=w end
            end
        end

        animTog.BackgroundColor3=S.anim and P.togOn or P.togOff
        camTog.BackgroundColor3=S.camR and P.togOn or P.togOff
        trailBtn.BackgroundColor3=S.trail and P.togOn or P.btn
        ghostBtn.BackgroundColor3=S.ghost and P.togOn or P.btn
        loopBtn.BackgroundColor3=S.loop and P.togOn or P.btn
        cfgBtn.BackgroundColor3=S.cfgOpen and P.togOn or P.btn
        trimFrontBtn.BackgroundColor3=(fc()>0 and S.frame>1) and P.btn or Color3.fromRGB(30,30,45)
        trimBtn.BackgroundColor3=(fc()>0 and S.frame>0 and S.frame<fc()) and P.btn or Color3.fromRGB(30,30,45)

        local canAdd=(S.mode=="paused" and fc()>0 and S.frame==fc() and not S.addFramePending)
        addFrameBtn.BackgroundColor3=canAdd and P.acc or (S.addFramePending and P.pause or Color3.fromRGB(30,30,45))
        addFrameBtn.TextColor3=(canAdd or S.addFramePending) and P.txt or P.dim
        addFrameBtn.Text=S.addFramePending and "..." or "+1F"

        if sfLbl then
            sfLbl.Text=tostring(sf()).."x"
            sfLbl.TextColor3=S.slowRec and P.slow or P.dim
        end

        local sl2=sl()
        if sl2 then
            local uc=sl2.undoStack and #sl2.undoStack or 0
            local rc=sl2.redoStack and #sl2.redoStack or 0
            undoInfoLbl.Text="U:"..uc.." R:"..rc
            undoBtn.BackgroundColor3=uc>0 and P.undoOn or P.undoOff
            undoBtn.TextColor3=uc>0 and P.txt or P.dim
            redoBtn.BackgroundColor3=rc>0 and P.undoOn or P.undoOff
            redoBtn.TextColor3=rc>0 and P.txt or P.dim
        end

        refreshBars()
    end)
end

local function onCharacter(char)
    if S.mode=="recording" or S.mode=="rec_paused" then stopRec() end
    if S.mode=="playing" or S.mode=="paused" then stopPlay() end
    clearAC()
    killGhost()
    removeSlowGravity()
    local hum=char:WaitForChild("Humanoid",10)
    if hum then
        S.sWS=hum.WalkSpeed
        S.sJP=hum.JumpPower
        hum.Died:Connect(function()
            if S.mode=="recording" or S.mode=="rec_paused" then stopRec() end
            if S.mode~="idle" then stopPlay() end
        end)
    end
    notify("Character ready",P.play)
end
